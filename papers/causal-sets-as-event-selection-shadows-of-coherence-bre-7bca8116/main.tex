\begin{abstract}
Causal set theory postulates discrete spacetime events endowed with a partial order,
while measurement theory postulates irreducible events associated with collapse or
irreversibility. We show that both structures arise as the same four-dimensional
shadow of coherence-selection events in Modal Triplet Theory (MTT). Selection events
are defined as irreversible basin transitions under noninvertible projection of the
coherent sector. We prove that these events generate a locally finite partial order,
yielding an effective causal set whose density, ordering, and approximate Lorentz
invariance are fixed by coherent-sector bottleneck data rather than by ad hoc
sprinkling assumptions. Deviations from Poisson statistics and exact Lorentz
invariance are predicted near coherence-breakdown regions. We relate these results
to ongoing work in causal set theory, continuous measurement, and collapse physics,
and establish cross-sector closure: the same bottleneck vector controls event density,
collapse thresholds, ultraviolet completion, and canonical geometric discreteness.
All statements are slab-local and admissibility-conditioned.
\end{abstract}

% ============================================================
\section{Introduction}
% ============================================================

Causal set theory proposes that spacetime is fundamentally discrete, consisting of
elementary events partially ordered by causality. Measurement theory and collapse
models, by contrast, treat events as irreducible outcomes of quantum dynamics, often
without spacetime ontology. These two notions of ``event'' are typically regarded as
independent.

In this paper we show that they are not independent. Within Modal Triplet Theory
(MTT), both causal set elements and measurement/collapse events arise as the same
objects: \emph{selection events} generated by coherence breakdown under noninvertible
projection. Spacetime discreteness, causal order, and irreversibility are therefore
different shadows of a single reduced-dynamical mechanism.

This work extends earlier results showing that causal sets arise as an effective
limit of MTT. Here we go further by identifying the physical meaning of causal set
elements as selection events and by deriving their density, ordering, and statistical
symmetries from coherent-sector data.

\section{Minimal Coherent-Sector Inputs}
% ============================================================

\begin{assumption}[Slab-local coherent regime]\label{ass:slab}
All results are asserted on bounded-geometry time slabs $\Omega=[0,T]\times\Sigma$
where:
\begin{enumerate}[label=(S\arabic*)]
\item the coherent projector is bounded on the controlled domain;
\item a uniform spectral gap $\lambda_\ast$ separates coherent and noncoherent modes;
\item admissible basins possess strictly positive stability margins away from boundaries.
\end{enumerate}
\end{assumption}

\begin{definition}[Bottleneck vector]\label{def:Theta}
The \emph{bottleneck vector} $\ThetaB$ is the finite set of coherent-sector data
controlling projection and stability:
\begin{enumerate}
\item the spectral gap $\lambda_\ast$;
\item projector norm and overlap data;
\item admissibility and stability-margin parameters.
\end{enumerate}
\end{definition}

\section{Selection Events in the Coherent Sector}
\label{sec:selection}
% ============================================================

\begin{definition}[Admissible basins]
Let $\mathcal{C}$ denote the coherent sector on the slab. Admissible configurations
lie in dynamically stable basins $\{\mathcal{B}_\alpha\}$ of the projected dynamics.
\end{definition}

\begin{definition}[Selection event]
A \emph{selection event} is the spacetime-localized occurrence of a trajectory exiting
an admissible basin $\mathcal{B}_\alpha$ and being irreversibly captured into a
different basin $\mathcal{B}_\beta$ under projection.
\end{definition}

\begin{proposition}[Irreversibility of selection events]
\label{prop:irreversible}
Selection events are irreversible on the slab: once capture into $\mathcal{B}_\beta$
occurs, return to $\mathcal{B}_\alpha$ with nonzero probability is excluded by
contractivity in $\mathcal{B}_\beta$.
\end{proposition}

\begin{proof}
Exit from $\mathcal{B}_\alpha$ occurs at loss of a stability margin. After capture into
$\mathcal{B}_\beta$, all unstable directions are contracted. Standard basin stability
arguments imply return probability below truncation tolerance. \qedhere
\end{proof}

% ============================================================
\section{Emergent Causal Order from Basin Selection}
\label{sec:order}
% ============================================================

\begin{definition}[Event dependence]
For selection events $e_1,e_2$, write $e_1\prec e_2$ if the basin selected by $e_1$ is a
necessary condition for the admissibility of the trajectory leading to $e_2$.
\end{definition}

\begin{theorem}[Emergent causal order]\label{thm:order}
The relation $\prec$ defines a partial order on selection events.
\end{theorem}

\begin{proof}
Transitivity follows from composition of basin dependencies. Antisymmetry follows
from irreversibility (Proposition~\ref{prop:irreversible}). Cycles would require
return to an exited basin, excluded by admissibility. \qedhere
\end{proof}

% ============================================================
\section{Effective Causal Sets as Four-Dimensional Shadows}
\label{sec:causets}
% ============================================================

\begin{definition}[Effective causal set]
The effective causal set $\mathcal{C}_{\mathrm{eff}}$ is the set of selection events
endowed with the partial order $\prec$.
\end{definition}

\begin{theorem}[Causal set axioms as shadows]\label{thm:causet}
$\mathcal{C}_{\mathrm{eff}}$ satisfies:
\begin{enumerate}[label=(C\arabic*)]
\item partial order;
\item local finiteness;
\item causal precedence.
\end{enumerate}
\end{theorem}

\begin{proof}
Partial order is Theorem~\ref{thm:order}. Local finiteness follows from gap-controlled
event density (Section~\ref{sec:density}). Causal precedence is encoded by basin
dependence. \qedhere
\end{proof}

% ============================================================
\section{Event Density, Lorentz Invariance, and Deviations}
\label{sec:density}
% ============================================================

\begin{proposition}[Event density]\label{prop:density}
The mean density of selection events per unit four-volume is fixed by the bottleneck
vector $\ThetaB$, in particular by the spectral gap and stability margins.
\end{proposition}

\begin{theorem}[Statistical Lorentz invariance]\label{thm:lorentz}
On scales large compared to the truncation scale, the distribution of selection events
is statistically Lorentz invariant.
\end{theorem}

\begin{proof}
The coherent sector is Lorentz covariant. Projection introduces no preferred frame.
Event statistics therefore inherit covariance in expectation values. \qedhere
\end{proof}

\begin{remark}
Near coherence-breakdown regions or strong measurement, event clustering and
Lorentz-violating correlations are predicted deviations from Poisson sprinkling.
\end{remark}

% ============================================================
\section{Relation to Causal Set Theory and Measurement Physics}
\label{sec:relation}
% ============================================================

Causal set theory postulates discreteness and Poisson sprinkling to preserve Lorentz
invariance. Here discreteness and approximate Lorentz invariance are derived, and the
sprinkling density is fixed by $\ThetaB$ rather than arbitrary.

In measurement theory, collapse events are postulated as irreducible. Here they are
identified with selection events that also generate causal order. Thus measurement
events and spacetime events are the same objects viewed through different diagnostics.

% ============================================================
\section{Cross-Sector Closure and Falsifiability}
\label{sec:closure}
% ============================================================

\begin{theorem}[Cross-sector closure]\label{thm:closure}
Within the coherent universality class on slabs, the following are fixed functions of
the same bottleneck vector $\ThetaB$:
\begin{enumerate}[label=(\roman*)]
\item event density and causal order;
\item collapse and measurement thresholds;
\item ultraviolet completion data;
\item canonical geometric discreteness.
\end{enumerate}
\end{theorem}

\begin{proof}
Each quantity depends only on projection-induced truncation, spectral gaps, and
stability margins encoded in $\ThetaB$. Independent tuning would require modifying
$\ThetaB$, affecting all sectors simultaneously. \qedhere
\end{proof}

\begin{corollary}[Falsifiability]
If any sector requires parameter values incompatible with $\ThetaB$ inferred from
another, the coherent-sector hypothesis is falsified.
\end{corollary}

\section{Conclusions}
% ============================================================

We have shown that causal sets and measurement events arise as the same shadow of
coherence-selection events under projection. Causal order, discreteness, and
irreversibility are unified as reduced-dynamical consequences of admissibility and
stability in the coherent sector.

Causal set discreteness is therefore effective rather than fundamental; Poisson
sprinkling is an approximation whose density and deviations are fixed by coherent
bottleneck data. Measurement collapse and spacetime events are identical objects
viewed through different lenses.

All results are slab-local and admissibility-conditioned. Together with independent
shadow constructions for collapse, measurement-induced transitions, ultraviolet
completion, and canonical geometry, this provides a tightly overconstrained validation
framework for coherence-based fundamental dynamics.
