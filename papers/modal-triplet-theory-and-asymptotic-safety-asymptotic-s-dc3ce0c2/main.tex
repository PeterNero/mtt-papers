\begin{abstract}
We establish a rigorous bridge between Modal Triplet Theory (MTT) and functional
renormalization group (FRG) formulations of quantum gravity, clarifying the precise
sense in which asymptotic safety (AS) arises as a controlled shadow of MTT's coherent
sector. Working on a bounded-geometry time slab and within the coherent universality
class, we define a background-covariant FRG effective average action $\Gamma_k$ and
a discrete renormalization-group (RG) step map $\mathcal{R}_b:\Gamma_k\mapsto\Gamma_{bk}$.
Using the spectral proper-time (SPT) Gaussian ultraviolet damping derived from the
coherent projector, we prove: (i) existence of a UV endpoint functional
$\Gamma_\infty=\lim_{k\to\infty}\Gamma_k$ in a Banach space of analytic local plus entire
form-factor actions equipped with a majorant-series norm; (ii) scheme stability of
$\Gamma_\infty$ under admissible regulator variations within the coherent universality
class; and (iii) compactness of the RG-step linearization at $\Gamma_\infty$, yielding a
finite-dimensional unstable manifold (the MTT-native analogue of ``finite relevant
directions''). As a corollary, fixed points observed in AS-style truncations are obtained
as shadows of $\Gamma_\infty$ under continuous truncation and canonical rescaling.
\end{abstract}

\section{Scope, standing inputs, and claim discipline}

\subsection{Standing inputs}
We work under the standing MTT modules $(G,S,\Pi,F,E)$: bounded geometry on a time
slab, uniform spectral gap separating coherent and noncoherent modes, bounded coherent
projector $\Pi_{\mathrm{coh}}$, well-posed smoothing flow, and analytic energy/interaction
structure. These are established in the MTT Foundation, Fixed Points I--VI, and the
Superset framework.

We restrict to the coherent universality class, i.e.\ admissible perturbations preserve
the coherent sector and its physical relevance, including robustness under RG-compatible
coarse-graining and controlled truncation.

The ultraviolet mechanism is the spectral proper-time (SPT) factorization of the coherent
projector, yielding Gaussian ultraviolet damping and a complete Bernstein/Stieltjes
structure in the TT graviton sector, compatible with BRST/BV symmetry.

\section{Slab geometry and bounded-geometry setting}

We work on a finite time slab $[0,T]\times\Sigma$ with bounded geometry: uniform bounds
on curvature and its derivatives and positive injectivity radius on the slab. All Sobolev,
heat-kernel, and parametrix estimates are uniform on this slab. This is the same setting
used in the constructive MTT--QG program.

\section{FRG representation and RG step map}

\subsection{Admissible regulator families}

\begin{definition}[Admissible regulator family]
A family $R_k=R_k(-\bar\nabla^2)$ is admissible if:
\begin{enumerate}
\item $R_k$ is background-covariant, depending on the background only via $-\bar\nabla^2$.
\item $R_k(z)\to k^2$ for $z\ll k^2$ and $R_k(z)\to0$ for $z\gg k^2$.
\item Regulator-modified Ward/Slavnov identities are of standard background-field type
and restore as $k\downarrow0$.
\item Regulator variations preserving (1)--(3) are bounded perturbations compatible with
coherent-sector stability and controlled truncation.
\end{enumerate}
\end{definition}

\subsection{Shell localization}

\begin{lemma}[Shell localization of admissible regulators]
\label{lem:shell}
For any admissible regulator family $R_k$, there exist constants $0<a_0<a_1$ such that
the operator $\partial_k R_k(-\bar\nabla^2)$ is spectrally localized (up to rapidly
decaying tails) to the band
\[
a_0 k^2 \;\le\; -\bar\nabla^2 \;\le\; a_1 k^2.
\]
\end{lemma}

\begin{proof}
This follows from the standard construction of FRG regulators as smooth functions of
$z/k^2$ with compact support or rapid decay of derivatives. The statement is verified
directly for all commonly used regulator shapes and is taken as part of admissibility.
\end{proof}

\subsection{Effective average action and RG step map}

Let $\Gamma_k$ denote the coherent-sector effective average action in background-field
FRG form. Define RG time $t=\log k$ and fix $b>1$.

\begin{definition}[RG step map]
The RG step map is
\[
\mathcal{R}_b:\Gamma_k\longmapsto\Gamma_{bk},
\]
the time-$\log b$ map of the FRG flow.
\end{definition}

\section{Action space and majorant-series norm}

\subsection{Analytic local and entire classes}

We define a Banach space $\mathcal{B}^{(s,R)}$ of admissible actions as the direct sum of:
\begin{itemize}
\item $\mathcal{A}_{\mathrm{loc}}^{(s,R)}$: local analytic functionals with factorial
bounds on Fr\'echet derivatives on the $H^s$-ball of radius $R$,
\item $\mathcal{A}_{\mathrm{ent}}^{(s,R)}$: entire form-factor completions (complete
Bernstein/Stieltjes functional calculus) with Gaussian ultraviolet domination in the TT sector.
\end{itemize}

\subsection{Majorant-series norm}

Fix $\rho>0$. For a functional $F$ smooth on the $H^s$-ball $B_R$, define
\[
\|F\|_{\mathrm{maj},\rho;R}
:=\sum_{n=0}^\infty \frac{\rho^n}{n!}\sup_{\Phi\in B_R}\|D^nF(\Phi)\|.
\]
For $\Gamma=V+W$ with $V\in\mathcal{A}_{\mathrm{loc}}$, $W\in\mathcal{A}_{\mathrm{ent}}$, set
\[
\|\Gamma\|_{\mathcal{B},\rho;R}
:=\inf_{\Gamma=V+W}
\bigl(\|V\|_{\mathrm{maj},\rho;R}+\|W\|_{\mathrm{ent},\rho;R}\bigr).
\]
This norm encodes analyticity via factorial bounds and is stable under sums, products,
and compositions required by FRG and RG-step operations.

\section{SPT-damped UV integrability}

\begin{assumption}[SPT-damped shell integrability]
\label{ass:spt}
There exist constants $C>0$, $c>0$, integer $m\ge0$, and $\tau_0>0$ such that for $k\ge k_0$,
\[
\|\partial_k\Gamma_k\|_{\mathcal{B},\rho;R}
\;\le\; C\,k^m e^{-c\tau_0 k^2}.
\]
\end{assumption}

This is the FRG analogue of Gaussian domination of graviton-containing graphs and follows
from SPT factorization and shell localization (Appendix~A).

\section{Existence of the UV endpoint functional}

\begin{theorem}[UV endpoint functional]
\label{thm:UVendpoint}
Under the standing inputs and Assumption~\ref{ass:spt}, the limit
\[
\Gamma_\infty:=\lim_{k\to\infty}\Gamma_k
\]
exists in $\mathcal{B}^{(s,R)}$ with respect to $\|\cdot\|_{\mathcal{B},\rho;R}$.
Moreover,
\[
\|\Gamma_\infty-\Gamma_k\|_{\mathcal{B},\rho;R}
\le \int_k^\infty C\,k'^m e^{-c\tau_0 k'^2}\,dk'.
\]
\end{theorem}

\begin{proof}
Write $\Gamma_k=\Gamma_{k_0}+\int_{k_0}^k\partial_{k'}\Gamma_{k'}\,dk'$.
Assumption~\ref{ass:spt} implies absolute integrability of the tail, hence Cauchy
convergence in the Banach norm.
\end{proof}

\section{Scheme stability}

\begin{theorem}[Scheme stability of $\Gamma_\infty$]
Let $R_k$ and $R'_k$ be admissible regulators generating flows $\Gamma_k$, $\Gamma'_k$
satisfying Assumption~\ref{ass:spt}. Let $\Gamma_\infty$, $\Gamma'_\infty$ be the
corresponding UV endpoints. Then $\Gamma_\infty$ and $\Gamma'_\infty$ lie in the same
coherent universality class. In particular there exists a bounded reparametrization
$\mathcal{U}$ and a controlled remainder $\mathcal{E}$ such that
\[
\Gamma'_\infty=\mathcal{U}\cdot\Gamma_\infty+\mathcal{E}.
\]
\end{theorem}

\section{Finite-dimensional unstable manifold}

\begin{theorem}[Finite unstable subspace]
Let $\mathcal{L}_b:=D\mathcal{R}_b|_{\Gamma_\infty}$. Then $\mathcal{L}_b$ decomposes as
$\mathcal{L}_b=\mathcal{S}_b+\mathcal{K}_b$ with $\mathcal{K}_b$ compact on
$\mathcal{B}^{(s,R)}$. Consequently, the unstable subspace
\[
\mathcal{U}_b:=\mathrm{span}\{v:\mathcal{L}_b v=\lambda v,\ |\lambda|>1\}
\]
is finite-dimensional.
\end{theorem}

\begin{proof}
Compactness of $\mathcal{K}_b$ follows from Gaussian-damped smoothing of the shell map
(Appendix~A). Riesz--Schauder theory then yields finite multiplicity of eigenvalues
outside the essential spectrum.
\end{proof}

\section{Asymptotic safety as a truncation shadow}

\begin{corollary}[AS fixed points as shadows]
Let $\mathcal{T}:\mathcal{B}^{(s,R)}\to\mathbb{R}^N$ be a continuous truncation.
Then $g(k):=\mathcal{T}(\Gamma_k)\to g(\infty):=\mathcal{T}(\Gamma_\infty)$.
After canonical rescaling to dimensionless couplings, any fixed point observed in the
truncation is a shadow of $\Gamma_\infty$, and the number of relevant directions is
bounded by $\dim\mathcal{U}_b$.
\end{corollary}

\section*{Appendix A.0: Majorant-Series Banach Norm}
\addcontentsline{toc}{section}{Appendix A.0: Majorant-Series Banach Norm}

\subsection*{A.0.1 Motivation}

We require a Banach space of actions that:
(i) encodes analyticity via factorial bounds,
(ii) is stable under differentiation, multiplication, and composition,
(iii) supports Cauchy convergence of the FRG flow,
(iv) is compatible with constructive QG methods.

The standard solution in constructive field theory is a majorant-series norm.

\subsection*{A.0.2 Analytic local functionals}

Fix $s>2$, $R>0$, and let
\[
B_R := \{\Phi : \|\Phi\|_{H^s([0,T]\times\Sigma)} \le R\}.
\]

For a Fréchet-smooth functional $F$ on $B_R$, define
\[
\|D^n F\|_R := \sup_{\Phi\in B_R} \|D^nF(\Phi)\|_{\mathrm{op}}.
\]

\begin{definition}[Majorant-series norm]
For $\rho>0$, define
\[
\|F\|_{\mathrm{maj},\rho;R}
:= \sum_{n=0}^\infty \frac{\rho^n}{n!}\,\|D^nF\|_R.
\]
\end{definition}

\begin{lemma}[Equivalence with factorial bounds]
$\|F\|_{\mathrm{maj},\rho;R}<\infty$ iff $F$ is analytic on $B_R$ with
\[
\|D^nF\|_R \le \|F\|_{\mathrm{maj},\rho;R}\,\rho^{-n} n!.
\]
Conversely, factorial bounds imply finiteness for $\rho<1/K$.
\end{lemma}

\subsection*{A.0.3 Entire form-factor sector}

Let $W$ be a functional defined by background-covariant kernels
(e.g.\ $F(-\bar\nabla^2)$ with $F$ complete Bernstein).
Define
\[
\|W\|_{\mathrm{ent},\rho;R}
:= \sum_{n=0}^\infty \frac{\rho^n}{n!}\|D^nW\|_R + \|W\|_{\mathrm{ker}},
\]
where $\|\cdot\|_{\mathrm{ker}}$ is the operator norm $H^s\to H^s$.

\subsection*{A.0.4 Full action space}

\begin{definition}[Banach space $\mathcal B^{(s,R)}$]
Let
\[
\mathcal B^{(s,R)} := \mathcal A_{\mathrm{loc}}^{(s,R)} \oplus
\mathcal A_{\mathrm{ent}}^{(s,R)},
\]
with norm
\[
\|\Gamma\|_{\mathcal B,\rho;R}
:= \inf_{\Gamma=V+W}
\bigl(\|V\|_{\mathrm{maj},\rho;R}+\|W\|_{\mathrm{ent},\rho;R}\bigr).
\]
\end{definition}

\begin{lemma}[Closure properties]
$\mathcal B^{(s,R)}$ is a Banach space, closed under sums, differentiation,
and FRG compositions. Analytic bounds are preserved with controlled constants.
\end{lemma}

\section*{Appendix A.1: Bounded-Geometry Slab and Heat-Kernel Bounds}
\addcontentsline{toc}{section}{Appendix A.1: Bounded-Geometry Slab and Heat-Kernel Bounds}

Let $[0,T]\times\Sigma$ be a finite time slab with bounded geometry.
For any Laplace-type operator $L$, the heat kernel $K_t(x,y)$ satisfies
\[
\|K_t(x,y)\|
\le C\,t^{-2}\exp\!\left(-\frac{d_{\bar g}(x,y)^2}{c\,t}\right),
\quad 0<t\le1.
\]

These estimates are uniform on the slab and allow parametrix transfer
between Fourier and configuration space.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Appendix A.2: SPT-Filtered Covariances}
\addcontentsline{toc}{section}{Appendix A.2: SPT-Filtered Covariances}

SPT factorization yields TT-sector covariances of the form
\[
C = f(L)\,(L+\lambda_*)^{-1},
\]
with $f$ completely monotone and supported on $[\tau_0,\infty)$.

In Fourier variables,
\[
\|\widehat C(p)\| \lesssim \frac{e^{-\tau_0|p|^2}}{|p|^2+\lambda_*}.
\]

\section*{Appendix A.3: Gaussian Damping Implies Sobolev Smoothing}
\addcontentsline{toc}{section}{Appendix A.3: Gaussian Damping Implies Sobolev Smoothing}

\begin{lemma}[Sobolev smoothing]
For any $\sigma\ge0$,
\[
C : H^s \to H^{s+\sigma}
\]
is bounded. The operator norm depends on $\tau_0,\lambda_*,\sigma$
and bounded-geometry constants.
\end{lemma}

\begin{proof}
The multiplier
\[
(1+|p|^2)^{\sigma/2}\frac{e^{-\tau_0|p|^2}}{|p|^2+\lambda_*}
\]
is bounded for all $\sigma$; transfer to the slab uses Appendix A.1.
\end{proof}

\section*{Appendix A.5: Compactness via Rellich on Slabs}
\addcontentsline{toc}{section}{Appendix A.5: Compactness via Rellich on Slabs}

\begin{lemma}[Rellich compactness]
On a bounded slab,
\[
H^{s+\sigma}\hookrightarrow H^s
\]
is compact for $\sigma>0$.
\end{lemma}

Combined with Appendix A.3, any smoothing operator
$H^s\to H^{s+\sigma}$ induces a compact map on $H^s$.

\section*{Appendix A.6: Quantitative Shell Contraction}
\addcontentsline{toc}{section}{Appendix A.6: Quantitative Shell Contraction}

\begin{lemma}[Shell localization]
$\partial_kR_k$ is spectrally supported in
$a_0k^2\le-\bar\nabla^2\le a_1k^2$.
\end{lemma}

\begin{lemma}[Quantitative contraction]
There exist constants $C'(b,\rho,R,s)$ and $c>0$ such that
\[
\|\mathcal K_{k\to bk}(\delta\Gamma)\|_{\mathcal B,\rho;R}
\le C'(b,\rho,R,s)\,k\,e^{-c\tau_0k^2}
\|\delta\Gamma\|_{\mathcal B,\rho;R}.
\]
\end{lemma}

\begin{proof}
Each TT propagator contributes $e^{-\tau_0|p|^2}$ on the shell,
yielding an overall $e^{-c\tau_0k^2}$ factor. Analytic bounds
absorb derivatives; integrate over $[k,bk]$.
\end{proof}

\section*{Appendix A.7: Spectral Consequences}
\addcontentsline{toc}{section}{Appendix A.7: Spectral Consequences}

The RG-step linearization decomposes as
\[
D\mathcal R_b = \mathcal S_b + \mathcal K_b,
\]
with $\mathcal K_b$ compact. By Riesz--Schauder theory,
only finitely many eigenvalues lie outside the unit disk.
Hence the unstable manifold is finite-dimensional.
