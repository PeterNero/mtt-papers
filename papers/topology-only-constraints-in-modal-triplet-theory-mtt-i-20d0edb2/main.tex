\begin{abstract}
We establish the Tier--1 ``low--hanging fruit'' predictions of the Modal Triplet Theory (MTT) rigorously, using only topology, group theory, and selection rules---\emph{no internal metric or harmonic integrals}. On a smooth oriented spin four--manifold $Y_4$, with three determinant line bundles obeying flux balance and pairwise twist sectors defined by the line bundles $L_{ij}$, we prove: (i) chiral family numbers come from a Dirac index formula; (ii) a topological criterion for Dirac vs.\ Majorana masses; (iii) absence of the SU(2) Witten global anomaly for three families; (iv) exact Standard--Model hypercharges from MTT difference charges; (v) full cancellation of all local gauge/gravitational anomalies; (vi) strong--CP relaxation via a central PQ--like symmetry with integer domain--wall number; (vii) forbiddance of many baryon/lepton number violating operators by line--bundle triviality; (viii) equality of photon and graviton wave speeds in the GR--compatible EFT; and (ix) the holonomy determinant/phase sum rule implied by the canonical trivialization $L_{12}\!\otimes L_{23}\!\otimes L_{31}\!\cong\!\mathbb C$.
\end{abstract}

\section{Setting and Notation}
Let $Y_4$ be a smooth, oriented, \emph{spin} four--manifold. The MTT Triplet data consist of three Hermitian line bundles
\[
\det B_1,\ \det B_2,\ \det B_3 \ \longrightarrow\ Y_4,\qquad
\alpha_i:=c_1(\det B_i)\in H^2(Y_4,\Z),
\]
obeying the \emph{flux balance}
\begin{equation}\label{eq:flux-balance}
\alpha_1+\alpha_2+\alpha_3=0.
\end{equation}
For each ordered pair $(i,j)$ define the twisting bundle
\[
L_{ij}:=\det B_i\otimes \det B_j,\qquad c_1(L_{ij})=\alpha_i+\alpha_j.
\]
All Tier--1 statements below are formulated purely in terms of these pairwise twist bundles,
their Chern classes, and standard representation bookkeeping on $Y^4$. No additional ``triple''
twisting line bundle is introduced: the only twisting data used throughout are the $L_{ij}$.


\begin{remark}[Canonical trivialization]
From \eqref{eq:flux-balance} one has
\[
c_1(L_{12}\otimes L_{23}\otimes L_{31})
=(\alpha_1+\alpha_2)+(\alpha_2+\alpha_3)+(\alpha_3+\alpha_1)
=2(\alpha_1+\alpha_2+\alpha_3)=0.
\]
Hence $L_{12}\otimes L_{23}\otimes L_{31}$ is topologically trivial. In what follows we
fix a unitary trivialization
\begin{equation}\label{eq:trivialization}
L_{12}\otimes L_{23}\otimes L_{31}\ \cong\ \mathbb C.
\end{equation}
which underlies the holonomy phase sum rule and the allowance of Yukawa--type trilinears.
\end{remark}

When discussing hypercharge, we use \emph{difference charges}
\begin{equation}\label{eq:diff-charges}
Y_{ij}:=y_i-y_j,\qquad y_1+y_2+y_3=0,
\end{equation}
so every $U(1)$ charge is a difference of node potentials $y_i$ with vanishing sum.

\section{Family Number by Index (Topology Only)}
Let $D$ be the \emph{chiral} Dirac operator on $Y_4$; for any Hermitian line bundle $L\!\to\!Y_4$ denote by $D\otimes L$ the twisted Dirac operator.

\begin{theorem}[Atiyah--Singer on $Y_4$]\label{thm:index}
For any Hermitian line bundle $L\to Y_4$,
\[
\mathrm{Index}(D\otimes L)\;=\;\int_{Y_4}\Big(\frac12\,c_1(L)^2-\frac1{24}\,p_1(TY_4)\Big)\ \in\ \Z.
\]
Hence for each bifundamental sector $(i,j)$,
\[
N_{ij}:=\mathrm{Index}(D\otimes L_{ij})=\int_{Y_4}\Big(\frac12\,c_1(L_{ij})^2-\frac1{24}\,p_1(TY_4)\Big).
\]
\end{theorem}

\begin{proof}
The Atiyah--Singer formula gives $\mathrm{Index}(D\otimes L)=\int_{Y_4}\hat A(TY_4)\,\mathrm{ch}(L)$. On a $4$--manifold, $\hat A=1-\frac{p_1}{24}$ and $\mathrm{ch}(L)=1+c_1+\frac12 c_1^2$. Taking the degree--$4$ component yields the stated expression.
\end{proof}

\begin{corollary}[Three families on $K3$]\label{cor:K3}
For $Y_4=K3$, $\int p_1=-48$ so $-\!\int \frac{p_1}{24}=2$. If $c_1(L_{ij})^2=2$, then
\(
N_{ij}=\frac12\cdot 2 + 2=3.
\)
\end{corollary}

\section{Dirac vs.\ Majorana: a Topological Criterion}
\begin{theorem}[Gauge--invariant Majorana mass]\label{thm:majorana}
Let $\Psi$ be a left--handed Weyl fermion valued in a line bundle $L\to Y_4$. A local Majorana mass term
\(
\frac12 m\,\Psi^T C\,\Psi+\mathrm{h.c.}
\)
is gauge invariant if and only if
\begin{equation}\label{eq:Maj-criterion}
L^{\otimes 2}\ \cong\ \mathbb C\qquad\Longleftrightarrow\qquad 2\,c_1(L)=0\ \text{ in }H^2(Y_4,\Z).
\end{equation}
If $H^2(Y_4,\Z)$ has no $2$--torsion (e.g.\ $K3$), \eqref{eq:Maj-criterion} is equivalent to $c_1(L)=0$ (hence $L\cong \overline L$).
\end{theorem}

\begin{proof}
The Majorana bilinear is a section of $L\otimes L$; gauge invariance demands $L^{\otimes 2}$ be trivial. In cohomology this is $2c_1(L)=0$. If $H^2$ is torsion--free, $2c_1=0$ implies $c_1=0$.
\end{proof}

\begin{remark}
Applied to the neutrino sector $(2,3)$, Majorana masses are allowed iff $2c_1(L_{23})=0$; on torsion--free $H^2$ this reduces to $c_1(L_{23})=0$. Otherwise light neutrinos are Dirac.
\end{remark}

\section{SU(2) Witten Global Anomaly: Parity Test}
\begin{theorem}[Absence of SU(2) global anomaly for three families]\label{thm:Witten}
The SU(2) global anomaly is absent if and only if the total number of left--handed SU(2) doublets is even. In the SM per family, $Q_L:(\mathbf 3,\mathbf 2)$ and $L_L:(\mathbf 1,\mathbf 2)$ give $3+1=4$ doublets (even). For three families this is $12$ (even); hence no global SU(2) anomaly.
\end{theorem}

\begin{proof}
Witten's anomaly is the mod--$2$ index classified by $\pi_4(SU(2))=\Z_2$ and equals the parity of SU(2) doublets counted as left Weyl fields. The stated counts are even.
\end{proof}

\begin{remark}[Parity via indices]
Writing $N_{12},N_{23}$ for family numbers in the quark and lepton SU(2) doublet sectors, the parity is $3N_{12}+N_{23}\ (\mathrm{mod}\ 2)$, which vanishes for $N_{12}=N_{23}=3$.
\end{remark}

\section{Exact SM Hypercharges from Difference Charges}
Let $Y_{ij}=y_i-y_j$ with $y_1+y_2+y_3=0$. Choose the identification
\[
Y(Q_L)=y_1-y_2,\quad Y(u_R)=y_1-y_3,\quad Y(L)=y_3-y_2,
\]
and use conjugation (sign flip) for $d_R,e_R$ as needed.

\begin{theorem}[Unique solution reproducing SM charges]\label{thm:hypercharge}
The system
\[
y_1-y_2=\tfrac{1}{6},\qquad y_1-y_3=\tfrac{2}{3},\qquad y_3-y_2=-\tfrac{1}{2},\qquad y_1+y_2+y_3=0
\]
has the unique solution
\[
y_1=\frac{5}{18},\qquad y_2=\frac{1}{9},\qquad y_3=-\frac{7}{18}.
\]
This yields
\[
Y(Q_L)=\tfrac{1}{6},\quad Y(u_R)=\tfrac{2}{3},\quad Y(d_R)=-\tfrac{1}{3},\quad
Y(L)=-\tfrac{1}{2},\quad Y(e_R)=-1,\quad Y(\nu_R)=0,
\]
i.e.\ precisely the SM hypercharges.
The overall normalization is fixed by choosing the smallest nonzero hypercharge unit
to be $1/6$, matching the conventional Standard Model normalization.

\end{theorem}

\begin{proof}
Subtract the third equation from the first to check consistency with the second. Solve the three linear equations and verify the sum constraint; then apply conjugation for $d_R,e_R$.
\end{proof}

\section{All Local Anomalies Cancel (Per Family)}
We compute with \emph{left--handed} Weyl fields only; right--handed SM fermions are represented as left--handed conjugates with opposite hypercharge.

\begin{theorem}[Anomaly cancellation]\label{thm:anomalies}
With the hypercharges of Theorem~\ref{thm:hypercharge}, per family one has
\[
[SU(3)]^2U(1)_Y = 0,\quad [SU(2)]^2U(1)_Y = 0,\quad [U(1)_Y]^3=0,\quad \mathrm{grav}^2 U(1)_Y=0.
\]
\end{theorem}

\begin{proof}
\textbf{(i) $[SU(3)]^2U(1)$:} Sum $Y$ over left Weyl triplets (the Dynkin index $T(\mathbf 3)$ is common and factors out):
\[
2\,Y(Q_L)+Y(u_L^c)+Y(d_L^c)=2\cdot\tfrac{1}{6}+(-\tfrac{2}{3})+(\tfrac{1}{3})=0.
\]
\textbf{(ii) $[SU(2)]^2U(1)$:} Sum $Y$ over doublets:
\[
3\,Y(Q_L)+Y(L)=3\cdot\tfrac{1}{6}+(-\tfrac{1}{2})=\tfrac{1}{2}-\tfrac{1}{2}=0.
\]
\textbf{(iii) $\mathrm{grav}^2 U(1)$:} Sum $Y$ over all left Weyl fields with multiplicities:
\[
6\cdot \tfrac{1}{6} + 3\cdot(-\tfrac{2}{3}) + 3\cdot(\tfrac{1}{3}) + 2\cdot(-\tfrac{1}{2}) + 1\cdot(+1)
=1-2+1-1+1=0.
\]
\textbf{(iv) $[U(1)]^3$:} Sum $Y^3$ with multiplicities:
\[
6\Big(\tfrac{1}{6}\Big)^3 + 3\Big(-\tfrac{2}{3}\Big)^3 + 3\Big(\tfrac{1}{3}\Big)^3 + 2\Big(-\tfrac{1}{2}\Big)^3 + 1\cdot(1)^3
=\frac{1}{36}-\frac{8}{9}+\frac{1}{9}-\frac{1}{4}+1=0.
\]
All vanish.
\end{proof}

\section{Strong--CP Relaxation and Domain--Wall Number}
\begin{theorem}[Central PQ--like symmetry $\Rightarrow \theta_{\rm QCD}\to 0$]\label{thm:PQ}
Suppose a global $U(1)_{\rm cen}$ with Goldstone mode $a/f_a$ has a QCD anomaly
\(
\frac{a}{f_a}\,\frac{g_s^2}{16\pi^2}\mathrm{tr}(G\tilde G)
\)
with integer coefficient $N_{\rm DW}\neq 0$. Then the vacuum relaxes to $\theta_{\rm eff}=\theta_{\rm QCD}+a/f_a=0$, i.e.\ $\theta_{\rm QCD}$ is dynamically canceled.
\end{theorem}

\begin{proof}
Standard Peccei--Quinn mechanism: the anomalous axial rotation shifts $\theta_{\rm QCD}$ by $a/f_a$; the nonperturbative effective potential $V\propto \cos(\theta_{\rm eff})$ is minimized at $\theta_{\rm eff}=0$.
\end{proof}

\begin{proposition}[Domain--wall number is group--theoretic]\label{prop:NDW}
The domain--wall number equals the color anomaly coefficient
\[
N_{\rm DW}=\sum_{\text{colored Weyl }\psi} 2\,T(R_\psi)\,Q_{\rm cen}(\psi)\ \in\ \Z,
\]
fixed by representation content and central charges (no geometry). One may choose charges with $N_{\rm DW}=1$.
\end{proposition}

\section{Operator Selection: Forbiddance by Line--Bundle Triviality}
A local gauge invariant built from fields $\Phi_a$ in sectors $(i_a,j_a)$ lives in
\[
\bigotimes_{a} L_{i_a j_a}.
\]
It is gauge invariant only if the tensor product is \emph{trivial}.

\begin{theorem}[Generic forbiddance of many B/L--violating terms]\label{thm:forbid}
Any trilinear composed of three identical bifundamentals requires
\(
L_{ij}^{\otimes 3}\cong \mathbb C\ \Leftrightarrow\ 3\,c_1(L_{ij})=0.
\)
On manifolds with torsion--free $H^2(Y_4,\Z)$ this forces $c_1(L_{ij})=0$; generically false. Hence such operators are forbidden. Conversely, the identity \eqref{eq:trivialization} ensures the \emph{Yukawa--type} trilinear with one field from each pair $(12),(23),(31)$ is allowed.
\end{theorem}

\begin{proof}
Triviality of a tensor product of line bundles is equivalent to the vanishing of the sum of their first Chern classes. Apply this to $L_{ij}^{\otimes 3}$. The Yukawa allowance follows because $c_1(L_{12}){+}c_1(L_{23}){+}c_1(L_{31})=0$.
\end{proof}

\section{Equality of Wave Speeds $c_{\rm em}=c_{\rm grav}$ in the EFT}
\begin{theorem}[Common characteristic cone]\label{thm:c-equals-c}
In the GR--compatible EFT with Einstein--Hilbert gravity and minimally coupled Maxwell/Yang--Mills, the principal symbols of the gauge--fixed linearized field operators are
\[
p(\xi)=g^{\mu\nu}\xi_\mu\xi_\nu\,\mathbb 1,
\]
so photons and gravitons propagate on the same null cone of $g$: $c_{\rm em}=c_{\rm grav}$.
\end{theorem}

\begin{proof}
Maxwell in Lorenz gauge: $(\square_g A_\nu - R_\nu{}^\mu A_\mu)=0$ has principal symbol $g^{\mu\nu}\xi_\mu\xi_\nu\,\delta^\alpha{}_\beta$. Linearized Einstein in de Donder gauge leads to the Lichnerowicz operator with principal part $\square_g h_{\mu\nu}$, with the same principal symbol. Thus the characteristic sets coincide.
\end{proof}

\section{Holonomy Determinant/Phase Sum Rule}
\begin{theorem}[Phase sum rule]\label{thm:phase-sum}
Let $U_{ij}(\gamma)$ be the parallel--transport (holonomy) on $L_{ij}$ along a closed loop $\gamma\subset Y_4$. Then
\[
\det U_{12}(\gamma)\cdot \det U_{23}(\gamma)\cdot \det U_{31}(\gamma)=1,
\]
equivalently,
\(
\arg\det U_{12}+\arg\det U_{23}+\arg\det U_{31}\in 2\pi\Z.
\)
\end{theorem}

\begin{proof}
Equip each $L_{ij}$ with its unitary Chern connection $A_{ij}$. The induced connection on the tensor product is $A_{12}{+}A_{23}{+}A_{31}$. By \eqref{eq:trivialization} this lives on the trivial bundle and admits a covariantly constant nonzero section, hence $A_{12}{+}A_{23}{+}A_{31}=d\chi$ for some gauge function $\chi$. Integrating around $\gamma$ gives $\exp\big(i\oint_\gamma(A_{12}+A_{23}+A_{31})\big)=\exp(i\oint_\gamma d\chi)=1$.
\end{proof}


\section{One--Loop RG Sign Structure (Representation--Only)}

\begin{proposition}[Qualitative one--loop flow (sign is representation--only)]\label{prop:RG-signs}
Fix the beta--function convention
\[
\beta(g)\;:=\;\mu\,\frac{dg}{d\mu}
\;=\;-\frac{g^3}{16\pi^2}\,b_0 \;+\; O(g^5).
\]
For a gauge factor $G$ with adjoint Casimir $C_A$, Weyl fermions in reps $R_f$
and complex scalars in reps $R_s$, the one--loop coefficient is
\[
b_0 \;=\; \frac{11}{3}C_A
\;-\;\frac{4}{3}\sum_f T(R_f)
\;-\;\frac{1}{6}\sum_s T(R_s).
\]
In this convention, asymptotic freedom is equivalent to $b_0>0$.
For Standard Model field content, one has $b_0^{SU(3)}>0$ and $b_0^{SU(2)}>0$ (asymptotically free),
while $b_0^{U(1)}<0$ (not asymptotically free; Landau pole in the UV / IR--free flow).
No geometry enters: only representation theory and multiplicities.
\end{proposition}

\begin{proof}
Standard one--loop calculation; the sign depends only on $C_A$, Dynkin indices $T(R)$,
and multiplicities of the matter representations.
\end{proof}

These results require no internal metric or harmonic analysis and form the rigorous base on which geometry--light (Tier~2) and calibratable (Tier~3) calculations can build.
