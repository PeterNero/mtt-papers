\title{\vspace{-1em}Modal Triplet Theory: From MTT to the Strominger (Heterotic Flux) System\\[0.25em]
\large{Selection of Non--K\"ahler SU(3) Geometry by a Fixed--Point Principle}}
\author{Peter Nero}
\date{September 7 2025}

\begin{document}
\maketitle

\begin{abstract}
We extend the Modal Triplet Theory (MTT) bridge from the torsion--free SU(3) (Calabi--Yau) slice to the \emph{non--K\"ahler, flux} slice governed by the Hull--Strominger system, and upgrade the result from admissibility to \emph{true selection by MTT}. On a compact complex threefold with trivial canonical bundle, conformally balanced metric, NS--NS three--form $\wh H$, and a Hermitian--Yang--Mills (HYM) bundle, we formulate \emph{twisted} Standing Assumptions (SA.F1--SA.F4) ensuring contractivity of the projected flow and the existence of a \emph{unique coherent fixed point}. We then construct a \emph{selection potential} $\Xi$ in the torsional SU(3) slice and prove: (i) its stationary points coincide with solutions of the Hull--Strominger system (with the Bismut/Hull connection on $TX$); (ii) under SA.F1--SA.F4, $\Xi$ is strictly convex near a Strominger solution, hence that solution is the \emph{unique local minimizer}; (iii) the MTT fixed point coincides with this minimizer and is globally attractive in the coherent sector. A reference implementation on the Fu--Yau class (torus bundles over K3 and K3--orbifolds) verifies twisted spectral gaps, bounded projectors, and commuting \emph{torsionful} Laplacians. Compactification ``choice'' is replaced by \emph{fixed--point selection} also in the non--K\"ahler regime; technically, this yields \emph{controlled truncations}, \emph{built--in decoherence}, and \emph{correlated low--energy data} in the presence of flux, fully parallel to the CY case.
\end{abstract}

\section{Introduction and Motivation}

\paragraph{MTT in brief.}
Modal Triplet Theory (MTT) posits a ten--dimensional modal geometry with three orthogonal internal bundles over a $4$D base and a joint harmonic projector defining the coherent sector. A dissipative flow $\Phi_t$ on field space, composed with the projector, yields a contractive map whose unique fixed point $\Psi^*$ projects to observed low--energy physics. The Calabi--Yau (CY) slice reproduces standard string backgrounds and effective dynamics while adding analytic control (spectral gaps, bounded projectors, commuting modal Laplacians) \cite{NeroMTTMaster2025, NeroString2025}.

\paragraph{Aim.}
We generalize the CY construction to the heterotic \emph{flux} regime: compact complex threefolds with trivial canonical bundle, non--K\"ahler (conformally balanced) metrics, nonzero flux and HYM bundles, subject to the \emph{Hull--Strominger system} \cite{Strominger1986,Bismut1989}. Beyond \emph{compliance}, we prove \emph{selection}: we build a torsional SU(3) \emph{selection potential} $\Xi$ and show that its unique local minimizer (under SA.F) is exactly the Strominger solution realized by the MTT fixed point.

\paragraph{Key corrections vs.\ prior draft.}
(i) We use the \emph{gauge--invariant Green--Schwarz 3--form} $\wh H$ (not a globally exact $H$), with $\dd\wh H=\frac{\alpha'}{4}\big(\Tr F\wedge F-\Tr R^+\wedge R^+\big)$. (ii) We endow $\Xi$ with \(|\wh H|^2\) and the Chern--Simons couplings, ensuring global gauge invariance (gerbe picture). (iii) We give full proofs (twisted gap, bounded projector, commuting torsionful blocks, Lyapunov descent, strict convexity, fixed point $=$ minimizer).

\paragraph{Contributions.}
\begin{itemize}[leftmargin=1.15em]
  \item Twisted Standing Assumptions (SA.F1--SA.F4) and a \emph{Flux Fixed--Point Theorem}. 
  \item A \emph{selection potential} $\Xi$ for $(g,J,\Omega,\Phi;B,A)$ using $\wh H$; Euler--Lagrange equations $\Leftrightarrow$ Hull--Strominger with $R^+$.
  \item \emph{Strict convexity} of $\Xi$ near a Strominger solution $\Rightarrow$ \emph{unique local minimum}; equality with the MTT fixed point and global attraction in the coherent sector.
  \item Fu--Yau implementation: admissibility checks and consistency.
  \item (Case study) Fu--Yau class (Sec.~\ref{sec:FuYau}); see also \emph{Appendix~\ref{app:iwasawa}} for a compact nilmanifold (Iwasawa) realization.

\end{itemize}

\paragraph{Organisation.}
Section~\ref{sec:Prelim} fixes function spaces, gauges, and $\wh H$. Section~\ref{sec:Strominger} recalls SU(3) torsion and the Strominger system. Section~\ref{sec:TwistedSA} proves SA.F1--SA.F4 and the fixed--point theorem. Section~\ref{sec:Selection} constructs $\Xi$ and proves selection. Sections~\ref{sec:WS}--\ref{sec:FuYau} discuss worldsheet $\beta=0$, HYM/Bianchi, and the Fu--Yau case. Appendices provide operator estimates, linearization, and commuting--block details.

\section{Preliminaries: Function Spaces, Gauges, and Green--Schwarz 3--form}
\label{sec:Prelim}

\subsection{Function spaces and bounded geometry}
All manifolds are smooth, compact, without boundary. Metrics, connections, and forms live in Sobolev spaces $H^s$ for $s$ large (or $C^{k,\alpha}$ H\"older spaces), with \emph{bounded geometry} hypotheses: uniform bounds on curvature and a positive injectivity radius. Norms $\|\cdot\|_{H^s}$, $\|\cdot\|_{C^{k,\alpha}}$ are taken with respect to a fixed background.

\subsection{Gauge choices and slices}
We work modulo diffeomorphisms and gauge:
\begin{itemize}[leftmargin=1.2em]
\item \emph{Metric}: DeTurck gauge for $g$ (fix a background connection to eliminate diffeo freedom in elliptic variations).
\item \emph{Gauge field}: unitary Coulomb gauge for $A$ (with respect to $g$) to fix $\mathcal{G}$.
\item \emph{B--field}: $B$ is a \emph{bundle 2--gerbe} connection; we work in local potentials and fix a gerbe gauge slice (Coulomb--type) consistent with large gauge transformations.
\end{itemize}

\subsection{Green--Schwarz 3--form and Bismut connection}
Define the Chern--Simons 3--forms
\[
\omega_3(A)=\Tr\!\big(A\wedge \dd A+\tfrac{2}{3}A\wedge A\wedge A\big),\quad
\omega_3(\omega^+)=\Tr\!\big(\omega^+\wedge \dd \omega^+ +\tfrac{2}{3}\omega^+\wedge \omega^+\wedge \omega^+\big),
\]
with $\omega^+$ the Bismut connection on $TX$ and curvature $R^+$. The \emph{gauge--invariant} Green--Schwarz 3--form is
\begin{equation}\label{eq:GHat}
\wh H \;=\; \dd B\;-\;\frac{\alpha'}{4}\,\big(\omega_3(A)-\omega_3(\omega^+)\big),\qquad
\dd \wh H \;=\; \frac{\alpha'}{4}\,\big(\Tr F\wedge F-\Tr R^+\wedge R^+\big).
\end{equation}
Throughout, the tangent--bundle connection in the Bianchi identity is $R^+$ (Bismut/Hull choice compatible with heterotic SUSY)~\cite{Strominger1986,Bismut1989}.

\subsection*{Gerbe covariance of $\wh H$}
The $B$-field is a Deligne 2-gerbe connection with connective structure. Local variations $\delta B$ are taken within a fixed differential cohomology class consistent with large gauge transformations; $\wh H$ is the globally defined curvature. The functional $\Xi$ depends on $B$ only via $\wh H$ and Chern--Simons forms, hence is gauge invariant. Variations are implemented by refining open covers and using a partition-of-unity argument; boundary terms cancel by the Bianchi constraint enforced by $K$.

\subsection*{Auxiliary multiplier for the Green–Schwarz definition}
We treat $\widehat H$ as an independent 3\textendash form variable and impose its Green--Schwarz definition 
by a 2\textendash form Lagrange multiplier $\Lambda\in\Omega^2(X)$:
\[
\widehat H \;=\; \dd B - \frac{\alpha'}{4}\big(\omega_3(A)-\omega_3(\omega^+)\big).
\]
This ensures global gauge invariance and makes the Euler--Lagrange variations with respect to $B$, $\widehat H$,
$A$, and $\omega^+$ fully consistent.


\section{SU(3)--Structure with Torsion and the Hull--Strominger System}
\label{sec:Strominger}

\subsection{SU(3) structure and torsion classes}
Let $(X^6,J,\Omega)$ be a complex threefold with Hermitian form $J$ and holomorphic $(3,0)$--form $\Omega$; intrinsic torsion decomposes into $W_1,\dots,W_5$. The CY case has $W_i=0$. In heterotic flux vacua typically $W_1=W_2=0$ (integrable complex structure), while $W_3$ and $W_4,W_5$ encode torsion and dilaton.

\subsection{Bismut connection and conformally balanced metrics}
The \emph{Bismut} connection $\nabla^+=\nabla^{\mathrm{LC}}+\tfrac12 T$ preserves $(g,J)$ with totally skew torsion $T=\wh H$ (torsionful connection); a Hermitian metric is \emph{conformally balanced} if $d(e^{-2\Phi}J^2)=0$ (Gauduchon class).

\subsection{Hull--Strominger system}
Given $(X,J,\Omega)$ with $K_X$ trivial, Hermitian metric $J$, dilaton $\Phi$, holomorphic bundle $(E,A)$, and $\wh H$ as in \eqref{eq:GHat},
\begin{align}
  & d\!\left(e^{-2\Phi}J\wedge J\right)=0, \quad \text{(conformally balanced)} \label{eq:balanced}\\
  & \wh H = i(\bar\partial-\partial)J\ -\ \frac{\alpha'}{4}\big(\omega_3(A)-\omega_3(\omega^+)\big), \label{eq:BismutTorsion}\\
  & F^{0,2}=0,\qquad J\lrcorner F=0, \quad \text{(HYM on $E$)} \label{eq:HYM}\\
  & \dd \wh H=\frac{\alpha'}{4}\Big(\mathrm{Tr}\,F\wedge F - \mathrm{Tr}\,R^+\!\wedge R^+\Big). \quad \text{(Bianchi / anomaly)} \label{eq:Bianchi}
\end{align}
These are necessary and sufficient for $4$D $\mathcal N{=}1$ heterotic SUSY at leading order in $\alpha'$. \cite{Strominger1986,Bismut1989,Fei2021}.

\section{Twisted Admissibility and the Flux Fixed--Point Theorem}
\label{sec:TwistedSA}

\subsection{Twisted Standing Assumptions (SA.F1--SA.F4)}
Let $X$ be compact complex with $K_X$ trivial, $(J,\Omega)$, Hermitian $g$, $\wh H$ as in \eqref{eq:GHat}, and holomorphic bundle $(E,A)$.

\medskip
\noindent\textbf{SA.F1 (Twisted spectral gap).}
Let $d_{\wh H}:=\dd+\wh H\wedge$ (or equivalently use Bismut Laplacians; see
also \cite{MathaiWu2011} for twisted de Rham complexes). The \emph{twisted Laplacian}
\[
  \Delta^{(\wh H)} \;=\; d_{\wh H}\delta_{\wh H}+\delta_{\wh H} d_{\wh H}
\]
is elliptic with discrete spectrum on compact $X$. On bounded--geometry families, there exists a uniform positive lower bound $\lambda_*^{(\wh H)}>0$ on the first nonzero eigenvalue.

\begin{lemma}[Uniform twisted gap]\label{lem:gap}
Assume bounded geometry (curvature bounds, injectivity radius $\ge \iota_0$) and $\|\wh H\|_{C^1}\le H_0$. Then
\[
\lambda_1\big(\Delta^{(\wh H)}\big)\ \ge\ \lambda_1(\Delta)\ -\ C(\iota_0,\|Rm\|_\infty)\,H_0,
\]
hence for $H_0$ small relative to the Cheeger/Buser constant one has $\lambda_*^{(\wh H)}\ge \tfrac12 \lambda_1(\Delta)>0$.
\end{lemma}

\begin{proof}
Write $\Delta^{(\wh H)}=\Delta+\mathsf{Q}_1(\nabla)+\mathsf{Q}_0$, where $\mathsf{Q}_1,\mathsf{Q}_0$ are first/zero order with operator norms bounded by $C H_0$. Kato--Rellich gives relative boundedness; min--max plus Cheeger/Buser estimate for $\Delta$  ~\cite{Cheeger1970,Buser1982} yields the claim.

\noindent\textit{Cheeger--Buser step.} For the Hodge Laplacian $\Delta$ on $k$-forms,
$\lambda_1(\Delta)\ge \tfrac{h^2}{4}$ with $h$ the (form) isoperimetric constant.
Hence $\lambda_1\big(\Delta^{(\wh H)}\big)\ge \tfrac{h^2}{4}-C\|\wh H\|_{C^1}$.

\noindent\textit{Constants.} All constants $C, c$ depend only on the bounded-geometry data 
$\big(\|Rm(g)\|_{C^0}, \operatorname{inj}(X,g)^{-1}\big)$ and the uniform $C^1$ bound on $\wh H$; they are uniform on the admissible family.
\end{proof}

\medskip
\noindent\textbf{SA.F2 (Bounded twisted projector).}
Let $\Pi^{(\wh H)}$ be the orthogonal projector onto twisted harmonic forms. Then $\Pi^{(\wh H)}:H^s\to H^s$ is bounded for all $s\in\R$, with norm controlled uniformly on the admissible family.

\begin{lemma}[Resolvent control $\Rightarrow$ bounded projector]\label{lem:proj}
For a small circle $\Gamma$ around $0$ in the resolvent set, 
\(
\Pi^{(\wh H)}=\frac{1}{2\pi i}\oint_\Gamma(\Delta^{(\wh H)}-z)^{-1}\dd z.
\)
Parameter--dependent elliptic estimates and Lemma~\ref{lem:gap} imply $\|(\Delta^{(\wh H)}-z)^{-1}\|_{H^s\to H^s}\le C_s$ uniformly; hence $\|\Pi^{(\wh H)}\|_{H^s\to H^s}\le C_s$.

\noindent\textit{Constants.} The bounds $C_s$ depend only on $s$ and the bounded-geometry data (as above) and on the uniform $C^1$ bound for $\wh H$; they are uniform over the admissible family.
\end{lemma}

\medskip
\noindent\textbf{SA.F3 (Commuting torsionful modal blocks).}
Assume $(g,\wh H)$ are block--diagonal with respect to the three internal bundles and that $\wh H=\sum_i \wh H_i$ with $\wh H_i$ supported on block $i$ (and any common $S^1$). Then the vertical twisted Laplacians commute:
\(
[\,\Delta^{(\wh H)}_{B_i},\,\Delta^{(\wh H)}_{B_j}\,]=0.
\)

\begin{lemma}[Block commutation]\label{lem:commute}
Under the above split, $d_{\wh H}=\sum_i (d_{(i)}+\wh H_i\wedge)$ and the terms act on disjoint coordinates. Hence $\Delta^{(\wh H)}=\sum_i \Delta^{(\wh H)}_{B_i}$ and the summands commute.
\end{lemma}

\medskip
\noindent\textbf{SA.F4 (Well--posedness and smoothing).}
The parabolic generator
\[
  \partial_t\Psi \;=\; -\Big(\sum_{i=1}^3 \kappa_i\,\Delta^{(\wh H)}_{B_i} + \varepsilon\,\Delta_Y\Big)\Psi \;-\; N(\Psi)
\]
is uniformly elliptic; the semigroup $\Phi_t$ is analytic and smoothing. On sublevels where $N$ is locally Lipschitz with constant $L$, the usual $M_1(t)e^{Lt}$ bounds hold (as in \cite{NeroMTTMaster2025}).

\subsection{Flux Fixed--Point Theorem}

\begin{theorem}[Flux fixed point \& uniqueness]\label{thm:FluxFixedPoint}
Assume SA.F1--SA.F4 and use $\nabla^+$ on $TX$. Then there exists $\tau>0$ such that
\[
  T_\tau \;=\; \Pi^{(\wh H)}_{\mathrm{coh}}\circ \Phi_\tau
\]
is a contraction on $(\mathrm{Ran}\,\Pi^{(\wh H)}_{\mathrm{coh}},\|\cdot\|_{H^1})$. Consequently, $T_\tau$ admits a \emph{unique} coherent fixed point $\Psi^*$, attracting all Picard iterates geometrically.
\end{theorem}

\begin{proof}
Combine Lemmas~\ref{lem:gap}--\ref{lem:proj} with standard parabolic smoothing to get $\|\Phi_\tau\|_{L^2\to H^1}\le M_1(\tau)e^{L\tau}$ and $\|\Pi^{(\wh H)}\|_{H^1\to H^1}\le C_\Pi$. Choose $\tau$ with $C_\Pi M_1(\tau)e^{L\tau}<1$; then Banach contraction applies.
\end{proof}

\section{Selection Potential and MTT Selection in the Flux Slice}
\label{sec:Selection}

\subsection{Configuration space and constraints}
Fix a compact complex threefold $X$ with $K_X\simeq \mathcal O_X$ and complex structure $J$ (thus $W_1=W_2=0$). Let
\[
\mathcal{C}=\big\{(g,\Phi,B;A):\ g\ \text{Hermitian on }(X,J),\ \Phi\in C^\infty,\ B\in\Omega^2,\ A\ \text{unitary on fixed holomorphic }E\big\},
\]
modulo diffeomorphisms, unitary gauge, and gerbe gauge. Define $\wh H$ by \eqref{eq:GHat}. Fix a topological sector (Chern data, cohomology class of $\wh H$ compatible with \eqref{eq:Bianchi}).

\subsection{Selection potential with Green--Schwarz 3--form}
In string frame, set
\begin{equation}\label{eq:XiDefFixed}
\begin{aligned}
\Xi[g,\Phi,B;A,\omega^+;\widehat H,K,\Lambda]\;=&\;
\int_X e^{-2\Phi}\Big(R(g)+4|\nabla\Phi|_g^2-\tfrac12|\widehat H|_g^2\Big)\,\mathrm{vol}_g \\
&\;+\;\frac{1}{2g_{10}^2}\int_X e^{-2\Phi}\,\Tr(F_A\wedge *F_A) \\
&\;+\;\int_X K\wedge\Big(\dd \widehat H-\tfrac{\alpha'}{4}(\Tr F_A\wedge F_A-\Tr R^+\wedge R^+)\Big)\\
&\;+\;\int_X \Lambda\wedge\Big(\widehat H-\dd B+\tfrac{\alpha'}{4}(\omega_3(A)-\omega_3(\omega^+))\Big)\\
&\;+\;\sum_{n,k}\frac{\delta_{n,k}}{2\,\gamma_{n,k}}\, .
\end{aligned}
\end{equation}

where $\omega^+$ is the Bismut connection, $R^+$ its curvature, and $\gamma_{n,k}=\kappa_{n,k}\lambda^{(\wh H)}_{n,k}-L$ (OU variance floor).

\begin{remark}
(i) $B$ enters only via $\wh H$ and Chern--Simons forms; large gauge transformations are respected. (ii) We impose the normalization $\int_X e^{-2\Phi}\vol_g=1$ to remove trivial rescalings.
\end{remark}

\subsection{Euler--Lagrange equations = Hull--Strominger}
\begin{theorem}[Euler--Lagrange $\Leftrightarrow$ Strominger]\label{thm:ELStromingerFixed}
Critical points of $\Xi$ on $\mathcal{C}\times\Omega^4$ obey the Hull--Strominger system \eqref{eq:balanced}--\eqref{eq:Bianchi} with $R^+$. Conversely, any smooth solution of \eqref{eq:balanced}--\eqref{eq:Bianchi} is a critical point of $\Xi$.
\end{theorem}

\begin{proof}
Varying $K$ enforces \eqref{eq:Bianchi}. Varying $B$ (hence $\wh H$) with fixed gerbe class yields $\dd(e^{-2\Phi}*\wh H)=0$, equivalent to \eqref{eq:balanced} on a complex threefold together with \eqref{eq:BismutTorsion}. Metric/dilaton variations give the $(G,B,\Phi)$ Euler--Lagrange system equivalent to worldsheet $\beta=0$ at leading $\alpha'$, which is known to be equivalent to \eqref{eq:balanced}--\eqref{eq:BismutTorsion} with $R^+$.~\cite{Polchinski1998,GSW1987}. Variation of $A$ in the holomorphic class gives HYM on a Gauduchon metric (Li--Yau), i.e.\ \eqref{eq:HYM}. The OU term is constant under deterministic variations. The converse follows by substitution.

\noindent\textit{Constants.} Coercivity and continuity constants depend only on bounded-geometry data and a uniform bound on $\|\wh H\|_{C^1}$; they are uniform on the admissible family.
\end{proof}

\subsection{Existence of minimizers}
\begin{proposition}[Direct method]\label{prop:existence}
In a fixed topological sector, under bounded geometry and SA.F1--SA.F2, $\Xi$ is bounded below and sequentially weakly lower semicontinuous on $\mathcal{C}$ modulo symmetries (with the normalization $\int e^{-2\Phi}\vol_g=1$). Hence $\Xi$ admits a minimizer.
\end{proposition}

\begin{proof}
Elliptic inequalities bound $\|R\|_{H^{-1}}$, $\|\nabla\Phi\|_{L^2}$, $\|\wh H\|_{L^2}$, and $\|F\|_{L^2}$ in terms of $\Xi$; SA.F1 and bounded geometry give compact embeddings (Rellich) modulo fixed gauges; lower semicontinuity holds by convexity of quadratic terms and continuity of the Chern--Simons constraint via $K$. The OU term is nonnegative and continuous.

\noindent\textit{Constants.} All constants are uniform on the admissible family and depend only on bounded-geometry data and the fixed Sobolev index used for compactness.
\end{proof}

\subsection{Strict convexity near a Strominger solution}
\begin{theorem}[Positive Hessian]\label{thm:convexity}
Let $(g_0,\Phi_0,B_0;A_0)$ solve \eqref{eq:balanced}--\eqref{eq:Bianchi} and satisfy SA.F1--SA.F4. Fix gauges as in Section~\ref{sec:Prelim}. Then there exists $c>0$ and a neighborhood $\mathcal U$ such that, for all $\mathsf u$ orthogonal to symmetry directions,
\[
\delta^2\Xi\big|_{(g_0,\Phi_0,B_0;A_0)}[\mathsf u,\mathsf u]\ \ge\ c\,\|\mathsf u\|{_{H^1}}^2.
\]
In particular, the solution is a \emph{unique local minimizer} in $\mathcal U$.
\end{theorem}

\begin{proof}
Linearize the system in fixed gauges; the principal symbol of the linearized operator is block--diagonal with entries the twisted Laplacians on the respective bundles (metric/dilaton via $\Delta^{(\wh H)}$ acting on symmetric 2--tensors and scalars; $B$ via $\Delta^{(\wh H)}$ on 2--forms; bundle via the Yang--Mills Laplacian). SA.F1 yields a uniform lower bound on the principal part; SA.F4 provides elliptic/smoothing control; SA.F2 controls projector pieces. If a kernel remains from true moduli, the OU term adds a positive quadratic form (weights $\gamma_{n,k}^{-1}$), lifting them. A G\r{a}rding inequality then gives the stated coercivity.

\noindent\textit{Constants.} The coercivity constant $c$ depends only on the bounded-geometry data, uniform $\|\wh H\|_{C^1}$ bounds, and the spectral gap $\lambda_*^{(\wh H)}$; it is uniform on the admissible family.
\end{proof}

\begin{remark}[OU variance term]
The OU term in $\Xi$ depends on the twisted eigenvalues $\lambda^{(\wh H)}_{n,k}$ via 
$\gamma_{n,k}=\kappa_{n,k}\lambda^{(\wh H)}_{n,k}-L$. 
In the Lyapunov inequalities we can drop this nonnegative term to obtain a lower bound, 
hence it never obstructs descent. 
In the Hessian analysis, its second variation contributes a nonnegative quadratic form 
along directions where $\delta\lambda^{(\wh H)}_{n,k}\neq 0$, and therefore can lift residual flat directions. 
Our coercivity estimate does not rely on it, but it strengthens positivity if present.
\end{remark}

\subsection{Lyapunov descent and equality with the fixed point}
\begin{proposition}[Strict Lyapunov for $\Phi_t$ and $T_\tau$]\label{prop:Lyap}
Along the flow $\Phi_t$ one has 
\[
\frac{\dd}{\dd t}\,\Xi(\Phi_t U)\ \le\ -\,c_0 \,\big\|\Pi^{(\wh H)}_{\mathrm{coh}}\nabla\Xi(\Phi_t U)\big\|_{H^{-1}}^2
\]
for some $c_0>0$. For the discrete map $T_\tau=\Pi^{(\wh H)}_{\mathrm{coh}}\circ\Phi_\tau$, there exists 
$\eta>0$ (independent of $U$) such that 
\[
\Xi(T_\tau U)\ \le\ \Xi(U)\ -\ \eta\,\big\|\Pi^{(\wh H)}_{\mathrm{coh}}\nabla\Xi(U)\big\|_{H^{-1}}^2.
\]
\end{proposition}

\begin{proof}
$\Phi_t$ is generated by a sectorial, maximally dissipative operator on the coherent sector; 
the chain rule plus elliptic regularity yields the stated differential inequality with $c_0$ from SA.F1–SA.F4. 
For $T_\tau$, use firm non-expansiveness of $\Pi^{(\wh H)}_{\mathrm{coh}}$ and the smoothing estimate 
$\|\Phi_\tau - (I-\tau \mathcal{A})\|_{H^{-1}\to H^{1}}\le C\tau^{3/2}$ with $\mathcal{A}$ the twisted elliptic part; 
a Krasnosel'ski\u{\i}–Mann argument (strong monotonicity of $\nabla\Xi$ on the slice) gives the discrete descent.

\noindent\textit{Constants.} $c_0,\eta$ depend only on the spectral gap $\lambda_*^{(\wh H)}$, Lipschitz constant $L$ of $N$ on the sublevel, and bounded-geometry data; they are uniform on the admissible family.
\end{proof}

\begin{theorem}[MTT selection]\label{thm:Selection}
Under SA.F1--SA.F4, in a fixed topological sector, the MTT fixed point $\Psi^*$ coincides with the \emph{unique local minimizer} of $\Xi$ (Theorems~\ref{thm:ELStromingerFixed} and \ref{thm:convexity}) and attracts all coherent iterates.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:FluxFixedPoint}, $T_\tau$ is a contraction with unique fixed point. By Proposition~\ref{prop:Lyap}, $\Xi$ strictly decreases along $T_\tau$ unless at a critical point; Theorem~\ref{thm:convexity} gives uniqueness of the local minimum. Hence the fixed point equals the minimizer and is globally attractive in the coherent sector.
\end{proof}

\section{Worldsheet $\sigma$--Model and $\beta=0$ in the Flux Slice}
\label{sec:WS}

In the MTT$\to$String dictionary \cite{NeroString2025}, the target fields $(G,B,\Phi)$ are pulled back from the coherent fixed point. The worldsheet $\sigma$--model (Polyakov action with Kalb--Ramond field) yields leading $\beta$--functions:
\[
\beta^{(G)}_{MN}\sim R_{MN}-\tfrac14 \wh H_{MPQ}\wh H_N{}^{PQ}+2\nabla_M\nabla_N\Phi,\quad
\beta^{(B)}_{MN}\sim -\tfrac12\nabla^P \wh H_{PMN}+\nabla^P\Phi\,\wh H_{PMN}.
\]
Vanishing $\beta$ at the fixed point---together with using $\nabla^+$ on $TX$---is equivalent to the Strominger system \eqref{eq:balanced}--\eqref{eq:Bianchi} at leading order in $\alpha'$ \cite{Strominger1986}. The heterotic gauge $\beta$ yields HYM on Gauduchon metrics (Li--Yau).\cite{LiYau1987}

\section{HYM on Gauduchon Metrics and Anomaly Cancellation}
\label{sec:HYM}

The Donaldson--Uhlenbeck--Yau correspondence extends beyond K\"ahler: on a compact complex manifold with a Gauduchon metric, slope--stable holomorphic bundles admit HYM connections \cite{LiYau1987,Buchdahl1988}. This furnishes \eqref{eq:HYM}. In torsional heterotic backgrounds the natural tangent connection entering \eqref{eq:Bianchi} is the torsionful Bismut/Hull connection $\nabla^+$; $R^+$ is the SUSY choice (equivalent to $R^-$ up to $\cO(\alpha')$ field redefinitions).

\section{Worked Case: The Fu--Yau Class}
\label{sec:FuYau}

\paragraph{Geometry.}
Let $\pi:X\to \mathrm{K3}$ be a principal $T^2$--bundle with complex structure and $K_X\simeq\cO_X$. Fu--Yau identified conformally balanced metrics with $\wh H=i(\bar\partial-\partial)J-\frac{\alpha'}{4}(\omega_3(A)-\omega_3(\omega^+))$ and solved \eqref{eq:balanced}--\eqref{eq:Bianchi} for suitable topological data. ~\cite{FuYau2008}.

For a complementary compact nilmanifold model with explicit left-invariant data, see Appendix~\ref{app:iwasawa}.


\paragraph{Admissibility checks (SA.F).}
\begin{itemize}[leftmargin=1.2em]
  \item \emph{Twisted spectral gap (SA.F1).} On compact $X$ with bounded geometry, $\Delta^{(\wh H)}$ is elliptic with discrete spectrum; Lemma~\ref{lem:gap} bounds $\lambda_1(\Delta^{(\wh H)})$ below uniformly on controlled families.
  \item \emph{Bounded twisted projector (SA.F2).} Lemma~\ref{lem:proj}.
  \item \emph{Commuting torsionful blocks (SA.F3).} Choose $(g,\wh H)$ split fiber/base; then Lemma~\ref{lem:commute} applies.
  \item \emph{Smoothing (SA.F4).} The twisted parabolic generator enjoys the same semigroup bounds as in CY \cite{NeroMTTMaster2025}.
\end{itemize}

\begin{proposition}[Model lower bound on $\lambda_1(\Delta^{(\wh H)})$ for Fu--Yau]
On a principal $T^2$-bundle over K3 with product-type ansatz $g=\epsilon^{-2}g_{T^2}\oplus g_{K3}$ and 
$\wh H$ supported in the base plus basic components, there exists $\epsilon_0>0$ and $c_*>0$ such that 
for all $0<\epsilon\le \epsilon_0$ one has $\lambda_1(\Delta^{(\wh H)})\ge c_*$. 
\end{proposition}

\begin{proof}
Use separation of variables: the first nonzero eigenvalue on the small fiber is $\gtrsim \epsilon^2$, 
while the base contribution is bounded below by the K3 isoperimetric constant; 
$\wh H$ enters as a first-order perturbation bounded uniformly in $\epsilon$, so Kato--Rellich preserves a uniform gap.
\end{proof}

\paragraph{HYM and Bianchi.}
Pick a slope--stable holomorphic bundle on the Gauduchon metric; by Li--Yau it carries an HYM connection. Use $\nabla^+$ on $TX$ in the Bianchi identity to complete a Strominger solution.

\paragraph{Conclusion.}
Fu--Yau manifolds furnish an \emph{admissible flux slice} for MTT; by Theorems~\ref{thm:convexity} and \ref{thm:Selection} they realize a \emph{selected} (unique local minimizing) Strominger geometry coincident with the MTT fixed point.

\section{Discussion and Outlook}

We have promoted the heterotic non--K\"ahler flux slice from \emph{compliance} to \emph{selection} inside MTT: twisted admissibility (SA.F1--SA.F4) grants a contraction/uniqueness theorem, while the Green--Schwarz--corrected selection potential $\Xi$ yields existence and \emph{strict convexity} at the Strominger solution, hence unique local minimization. Equality of minimizer and fixed point follows from Lyapunov descent. As in the CY corner, this provides selection, controlled truncations, built--in decoherence, and correlated low--energy data.

\paragraph{Scope in $\alpha'$.}
All target-space equations are matched to worldsheet $\beta$-functions at leading order in $\alpha'$. 
Higher-derivative corrections can be incorporated into $\Xi$ as additional local functionals; 
the contraction and Lyapunov parts of the proof depend only on sectoriality and bounded geometry 
and therefore persist qualitatively when such terms are perturbative.

\section{Elliptic and Resolvent Estimates for $\Delta^{(\wh H)}$}\label{app:elliptic}

Write $\Delta^{(\wh H)}=\Delta+\mathsf{Q}_1(\nabla)+\mathsf{Q}_0$ with $\|\mathsf{Q}_1\|+\|\mathsf{Q}_0\|\le C\|\wh H\|_{C^1}$. On compact bounded--geometry manifolds, parameter--dependent elliptic estimates give
\[
\|u\|_{H^{s+2}}\ \le\ C\Big(\|(\Delta^{(\wh H)}-z)u\|_{H^{s}} + (1+|z|)\|u\|_{H^{s}}\Big),
\]
uniform in $z$ on circles excluding the spectrum. Consequently, the resolvent $(\Delta^{(\wh H)}-z)^{-1}$ is bounded $H^s\to H^s$ uniformly on such circles, proving Lemma~\ref{lem:proj}. Lemma~\ref{lem:gap} follows from min--max and Cheeger/Buser for $\Delta$ plus relative boundedness of $\mathsf{Q}_1,\mathsf{Q}_0$.

\section{Linearization and Positive Hessian}\label{app:lin}
Linearize $\Xi$ at a Strominger background in fixed gauges. The second variation is
\[
\delta^2\Xi[\mathsf u,\mathsf u]\ =\ \langle \mathcal L \mathsf u,\ \mathsf u\rangle_{L^2} \ +\ \text{l.o.t.},
\]
where $\mathcal L$ is block--diagonal at leading order with entries the twisted Laplacians on the metric/dilaton ($\Delta^{(\wh H)}$ acting on symmetric 2--tensors and scalars), on $B$ ($\Delta^{(\wh H)}$ on 2--forms), and on the bundle ($\Delta_A$ acting on $\mathfrak{u}(E)$--valued 1--forms). SA.F1 yields $\langle \mathcal L \mathsf u,\mathsf u\rangle\ge c\|\mathsf u\|_{H^1}^2$ modulo symmetries; the OU variance term contributes $+\sum \gamma_{n,k}^{-1}|u_{n,k}|^2$, lifting residual moduli. This proves Theorem~\ref{thm:convexity}.

\section{Commuting Torsionful Blocks: Details}\label{app:commute}
If $g=\bigoplus_i g_i$ and $\wh H=\sum_i \wh H_i$ with $\wh H_i$ supported on block $i$ (and possibly a common $S^1$), then in block coordinates one has
\[
d_{\wh H}=\sum_i (d_{(i)}+\wh H_i\wedge),\qquad \delta_{\wh H}=\sum_i (\delta_{(i)}+\iota(\wh H_i)),
\]
and each summand acts only on block $i$. Hence $\Delta^{(\wh H)}=\sum_i \Delta^{(\wh H)}_{B_i}$ and $[\,\Delta^{(\wh H)}_{B_i},\Delta^{(\wh H)}_{B_j}\,]=0$.

\section{Detailed Variations in $(B,\widehat H,K,\Lambda)$}\label{app:BHVar}

We work at fixed topology and in the gauges of §\ref{sec:Prelim}. The only nontrivial variations from $\Xi$ in 
\eqref{eq:XiDefFixed} are in the $|\widehat H|^2$ term and the two constraint terms.

\paragraph{(i) Variation in $K$.}
\[
\delta_K \Xi \;=\; \int_X \delta K \wedge \Big(\dd \widehat H-\tfrac{\alpha'}{4}(\Tr F\wedge F-\Tr R^+\wedge R^+)\Big),
\]
so the Euler--Lagrange equation is the Bianchi identity
$\dd \widehat H=\tfrac{\alpha'}{4}(\Tr F\wedge F-\Tr R^+\wedge R^+)$.

\paragraph{(ii) Variation in $\Lambda$.}
\[
\delta_\Lambda \Xi \;=\; \int_X \delta\Lambda \wedge \Big(\widehat H-\dd B+\tfrac{\alpha'}{4}(\omega_3(A)-\omega_3(\omega^+))\Big),
\]
imposing the \emph{definition} of $\widehat H$ as the Green--Schwarz 3\textendash form:
$\widehat H=\dd B-\tfrac{\alpha'}{4}(\omega_3(A)-\omega_3(\omega^+))$.

\paragraph{(iii) Variation in $\widehat H$.}
Using $\delta |\widehat H|^2=2\langle \widehat H,\delta\widehat H\rangle$ and integrating by parts in the $K$ term,
\[
\delta_{\widehat H}\Xi \;=\; -\int_X e^{-2\Phi}\langle \widehat H,\delta\widehat H\rangle\,\mathrm{vol}_g
\;+\;\int_X K\wedge \dd(\delta\widehat H)\;+\;\int_X \Lambda\wedge \delta\widehat H .
\]
With $\dd(K\wedge \delta\widehat H)=\dd K\wedge \delta\widehat H - K\wedge \dd(\delta\widehat H)$ and no boundary,
this gives
\[
\delta_{\widehat H}\Xi \;=\; \int_X \big(-e^{-2\Phi}*\widehat H+\dd K+\Lambda\big)\wedge \delta\widehat H .
\]
Hence the equation of motion is
\begin{equation}\label{eq:EL1}
e^{-2\Phi}*\widehat H\;=\;\dd K+\Lambda .
\end{equation}

\paragraph{(iv) Variation in $B$.}
Only the $\Lambda$ term contributes:
\[
\delta_B\Xi \;=\; -\int_X \Lambda\wedge \dd(\delta B)\;=\;\int_X \dd\Lambda \wedge \delta B .
\]
Thus $\dd\Lambda=0$. We may absorb the closed form $\Lambda$ into $K$ (or set $\Lambda=0$ by choosing the
gauge slice for $K$), so \eqref{eq:EL1} becomes
\[
\dd\!\big(e^{-2\Phi}*\widehat H\big)\;=\;0,
\]
i.e. the weighted $B$--equation of motion.

\paragraph{(v) Variations in $A$ and $\omega^+$.}
The standard identities $\delta\omega_3(A)=2\Tr(\delta A\wedge F)+\dd\,\Tr(A\wedge \delta A)$ and the analogous one for 
$\omega^+$ show that the only bulk contribution from the constraints is
$-\tfrac{\alpha'}{2}\int_X K\wedge \Tr(\delta A\wedge F)$ and $+\tfrac{\alpha'}{2}\int_X K\wedge \Tr(\delta\omega^+\wedge R^+)$,
which combine with the Yang--Mills and Einstein terms. In the holomorphic class and for Gauduchon metrics, 
the gauge Euler--Lagrange equations reduce to the HYM conditions used in §\ref{sec:HYM}.

\subsection*{E.1\quad Background and SU(3) structure}
Let $X = \Gamma\backslash H_3(\mathbb{C})$ be the Iwasawa manifold with the standard left-invariant $(1,0)$-coframe
$(\omega_1,\omega_2,\omega_3)$ obeying $d\omega_1=d\omega_2=0$, $d\omega_3=\omega_1\wedge\omega_2$.
Set the Hermitian form and holomorphic volume form
\[
J=\frac{\ii}{2}\sum_{j=1}^3 \omega_j\wedge\bar\omega_j,\qquad \Omega=\omega_1\wedge\omega_2\wedge\omega_3.
\]
Then $X$ is complex parallelizable and $J$ is balanced ($d(J^2)=0$). With the Bismut connection $\nabla_+$ on $TX$,
the torsion is $T=H_b=\ii(\bar\partial-\partial)J$, consistent with the Green--Schwarz definition used in the main text.

\subsection*{E.2\quad Strominger data and the Bianchi identity}
Use $\nabla_+$ on $TX$ and take a left-invariant SU(3)-instanton $A$ on a holomorphic bundle $E\to X$
(explicit left-invariant abelian instantons exist on Iwasawa). Define the gauge-invariant Green--Schwarz three-form
\[
H_b = dB - \frac{\alpha'}{4}\big(\omega_3(A)-\omega_3(\omega_+)\big),\qquad
dH_b = \frac{\alpha'}{4}\big(\Tr F\wedge F - \Tr R_+\wedge R_+\big).
\]
Then, with $J$ balanced and $A$ of HYM type, the \emph{Hull--Strominger system}
$d(e^{-2\Phi}J\wedge J)=0$, $F^{0,2}=0$, $J\lrcorner F=0$, and the Bianchi identity above is satisfied.

\subsection*{E.3\quad Twisted admissibility (SA.F) on Iwasawa}
Let $\Delta(H_b)$ be the twisted Laplacian for $d_{H_b}=d+H_b\wedge$. On a fixed compact Iwasawa metric:
\begin{itemize}
\item \textbf{Twisted spectral gap (SA.F1).} $\lambda_1(\Delta(H_b))>0$, with a uniform lower bound on bounded-geometry families.

\item \textbf{Bounded twisted projector (SA.F2).} The twisted harmonic projector $\Pi(H_b):H^s\to H^s$ is bounded for all $s$,

\subsection*{E.4\quad Selection: minimizer $=$ MTT fixed point}
Let $\Xi$ be the selection potential of Sec.~5. By Theorem~5.2 (EL $\Leftrightarrow$ Strominger), any Iwasawa Strominger
background is a critical point. Near a left-invariant solution $(J_0,\Phi_0,B_0;A_0)$, the Hessian is strictly positive
modulo symmetries (Theorem~5.4), and $\Xi$ is a strict Lyapunov function for the coherent flow and the discrete map
$T_\tau$ (Proposition~5.6). Hence the coherent MTT fixed point exists, is unique, and \emph{equals} the unique local minimizer,
i.e.\ the Iwasawa Strominger solution (A02 Theorem~1). :contentReference[oaicite:14]{index=14} :contentReference[oaicite:15]{index=15}
